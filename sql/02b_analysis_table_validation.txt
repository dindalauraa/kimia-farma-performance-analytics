SELECT
  COUNT(*)                          AS total_rows,
  COUNT(DISTINCT transaction_id)    AS unique_trx,
  SUM(nett_sales)                   AS total_nett_sales,
  SUM(nett_profit)                  AS total_nett_profit,
  COUNTIF(branch_name IS NULL)      AS missing_branch,
  COUNTIF(product_name IS NULL)     AS missing_product,
  COUNTIF(nett_sales < 0)           AS negative_sales,
  MIN(date)                         AS first_date,
  MAX(date)                         AS last_date
FROM `rakamin-kf-analytics-511307.kimia_farma.tabel_analisa`;
